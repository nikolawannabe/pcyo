{{- /* A calendar file (.ics) for one class with every day it meets this year (partials/class-sessions.html), published at /calendars/-.ics, and the links to subscribe to it. Subscribing (not importing) means a family's calendar picks up date changes when the site is updated; Google checks about once a day. Takes one class from data/classes.yaml. Returns "" when the class has no start time or days yet, or has calendar: false; otherwise { ics: the file's address, google: "add to Google Calendar" link, webcal: subscribe link for other calendar apps } Times are in Pacific time. Make-up days are marked tentative; early-start days say so. */ -}} {{- $c := . -}} {{- $start := partial "clock-minutes.html" ($c.start | default "") -}} {{- $result := "" -}} {{- if and (ge $start 0) $c.days (ne $c.calendar false) -}} {{- $length := int ($c.length | default 60) -}} {{- $host := (urls.Parse site.BaseURL).Host -}} {{- $slug := urlize (printf "%s %s" $c.town $c.name) -}} {{- $where := printf "%s, WA" ($c.location | default $c.town) -}} {{- $page := (site.GetPage "/current-students").Permalink -}} {{- $events := slice -}} {{- range partial "class-sessions.html" $c -}} {{- $day := .date -}} {{- $from := $day.Add (time.ParseDuration (printf "%dm" $start)) -}} {{- $title := printf "%s (PCYO)" $c.name -}} {{- $note := printf "%s" .term -}} {{- if eq .kind "makeup" }}{{ $title = printf "%s: make-up day" $title }}{{ $note = printf "%s make-up day" .term }}{{ end -}} {{- if eq .kind "early" }}{{ $note = printf "%s, early start" .term }}{{ end -}} {{- with $c.teacher }}{{ $note = printf "%s · with %s" $note . }}{{ end -}} {{- $note = printf "%s · Days off and changes: %s" $note $page -}} {{- /* a day the class meets somewhere else (exported by the scheduler) */ -}} {{- $at := $where -}} {{- with .location }}{{ $at = printf "%s, WA" . }}{{ end -}} {{- $events = $events | append (dict "uid" (printf "%s-%s@%s" ($day.Format "20060102") $slug $host) "from" $from "to" ($from.Add (time.ParseDuration (printf "%dm" $length))) "title" $title "where" $at "note" $note "tentative" (eq .kind "makeup")) -}} {{- end -}} {{- $result = partial "ics-file.html" (dict "path" (printf "calendars/%s.ics" $slug) "name" (printf "%s (%s)" $c.name $c.town) "events" $events) -}} {{- end -}} {{- return $result -}}
