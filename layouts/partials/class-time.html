{{- /* A class's meeting time as text, built from days / start / length in data/classes.yaml. Usage: partial "class-time.html" (dict "c" $class "style" "short") short (tables, phone cards): "Mon & Wed 4:30–5:15pm", "Mondays 11:00–11:50am" long (class pages, Holiday Session page): "Mondays & Wednesdays, 4:30–5:15pm (45 minutes)" A class that only meets in optional terms (Holiday Session) also gets its dates, from the term dates minus days off: "Mondays 3:30pm · Nov 30, Dec 7, 14". */ -}} {{- $c := .c -}}{{- $long := eq .style "long" -}} {{- $full := dict "Mon" "Monday" "Tue" "Tuesday" "Wed" "Wednesday" "Thu" "Thursday" "Fri" "Friday" "Sat" "Saturday" "Sun" "Sunday" -}} {{- $days := $c.days | default slice -}} {{- /* days */ -}} {{- $dayText := "" -}} {{- if and (not $long) (gt (len $days) 1) -}} {{- $dayText = delimit $days " & " -}} {{- else -}} {{- $names := slice }}{{ range $days }}{{ $names = $names | append (printf "%ss" (index $full . | default .)) }}{{ end -}} {{- $dayText = delimit $names " & " -}} {{- end -}} {{- /* clock time: "4:30pm" -> minutes after midnight (partials/clock-minutes.html) */ -}} {{- $clock := "" -}} {{- with $c.start -}} {{- $s := partial "clock-minutes.html" . -}} {{- if ge $s 0 -}} {{- $sAp := cond (ge $s 720) "pm" "am" -}} {{- $sH := mod (div $s 60) 12 }}{{ if eq $sH 0 }}{{ $sH = 12 }}{{ end -}} {{- $sText := printf "%d:%02d" $sH (mod $s 60) -}} {{- with $c.length -}} {{- $e := mod (add $s (int .)) 1440 -}} {{- $eAp := cond (ge $e 720) "pm" "am" -}} {{- $eH := mod (div $e 60) 12 }}{{ if eq $eH 0 }}{{ $eH = 12 }}{{ end -}} {{- $eText := printf "%d:%02d%s" $eH (mod $e 60) $eAp -}} {{- if eq $sAp $eAp }}{{ $clock = printf "%s–%s" $sText $eText }}{{ else }}{{ $clock = printf "%s%s–%s" $sText $sAp $eText }}{{ end -}} {{- if $long }}{{ $clock = printf "%s (%d minutes)" $clock (int .) }}{{ end -}} {{- else -}} {{- $clock = printf "%s%s" $sText $sAp -}} {{- end -}} {{- else -}} {{- $clock = . -}} {{- end -}} {{- end -}} {{- $out := $dayText -}} {{- with $clock }}{{ $out = printf "%s%s%s" $out (cond $long ", " " ") . }}{{ end -}} {{- /* dates, for classes that only meet in optional terms */ -}} {{- $T := site.Data.terms -}} {{- $cterms := $c.terms | default (slice "fall" "winter" "spring") -}} {{- $onlyOptional := true -}} {{- range $cterms }}{{ $id := . }}{{ $t := index (where $T.terms "id" $id) 0 }}{{ if not $t.optional }}{{ $onlyOptional = false }}{{ end }}{{ end -}} {{- if and $onlyOptional $days -}} {{- $dates := slice }}{{ $prevMonth := "" -}} {{- range $cterms }}{{ $id := . }}{{ with index (where $T.terms "id" $id) 0 -}} {{- $ts := time .start }}{{ $n := int (div ((time .end).Sub $ts).Hours 24) -}} {{- range seq 0 $n }}{{ $d := $ts.AddDate 0 0 . -}} {{- if in $days ($d.Format "Mon") -}} {{- $off := false -}} {{- range $T.breaks }}{{ if and (not ($d.Before (time .start))) (not ($d.After (time .end))) (or (eq .who "all") (in ($c.teacher | default "") .who)) }}{{ $off = true }}{{ end }}{{ end -}} {{- if not $off -}} {{- if eq ($d.Format "Jan") $prevMonth }}{{ $dates = $dates | append ($d.Format "2") }}{{ else }}{{ $dates = $dates | append ($d.Format "Jan 2") }}{{ end -}} {{- $prevMonth = $d.Format "Jan" -}} {{- end -}} {{- end -}} {{- end -}} {{- end }}{{ end -}} {{- with $dates }}{{ $out = printf "%s · %s" $out (delimit . ", ") }}{{ end -}} {{- end -}} {{- return $out -}}
